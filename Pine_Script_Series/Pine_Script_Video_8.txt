//@version=6
strategy("Pine Script Video 8 - EMA", overlay = true)

longMessage = input.string(defval = "Long Message", title = "Long Message", group = "JSON SYNTAX INPUT")
shortMessage = input.string(defval = "Short Message", title = "Short Message", group = "JSON SYNTAX INPUT")

// EMA Settings
ema1Length = input.int(defval = 9, title = "EMA1 Length", inline = "EMA1", group = "EMA1 Settings")
ema1Source = input.source(defval = close, title = "EMA1 Source", inline = "EMA1", group = "EMA1 Settings")
ema1Color = input.color(defval = color.green, title = "EMA1 Color", inline = "EMA1", group = "EMA1 Settings")

ema1 = ta.ema(ema1Source, ema1Length)

plot(ema1, title = "EMA1", color = ema1Color, force_overlay = true)

// EMA2 Settings
ema2Length = input.int(defval = 26, title = "EMA2 Length", inline = "EMA2", group = "EMA2 Settings")
ema2Source = input.source(defval = close, title = "EMA2 Source", inline = "EMA2", group = "EMA2 Settings")
ema2Color = input.color(defval = color.orange, title = "EMA2 Color", inline = "EMA2", group = "EMA2 Settings")

ema2 = ta.ema(ema2Source, ema2Length)

plot(ema2, title = "EMA2", color = ema2Color, force_overlay = true)

longCondition = ta.crossover(ema1, ema2)
shortCondition = ta.crossunder(ema1, ema2)

if longCondition and barstate.isconfirmed
    strategy.entry("BUY", direction = strategy.long, alert_message = longMessage)

if shortCondition and barstate.isconfirmed
    strategy.entry("SELL", direction = strategy.short, alert_message = shortMessage)
