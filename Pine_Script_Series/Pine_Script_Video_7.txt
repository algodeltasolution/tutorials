//@version=6
strategy("EMA Strategy", overlay = true)

// EMA Settings
ema1Length = input.int(defval = 3, title = "EMA1 Length", inline = "EMA1", group = "EMA1 Settings")
ema1Source = input.source(defval = close, title = "EMA1 Source", inline = "EMA1", group = "EMA1 Settings")
ema1Color = input.color(defval = color.green, title = "EMA1 Color", inline = "EMA1", group = "EMA1 Settings")

ema1 = ta.ema(ema1Source, ema1Length)

plot(ema1, title = "EMA1", color = ema1Color, force_overlay = true)

// EMA2 Settings
ema2Length = input.int(defval = 5, title = "EMA2 Length", inline = "EMA2", group = "EMA2 Settings")
ema2Source = input.source(defval = close, title = "EMA2 Source", inline = "EMA2", group = "EMA2 Settings")
ema2Color = input.color(defval = color.orange, title = "EMA2 Color", inline = "EMA2", group = "EMA2 Settings")

ema2 = ta.ema(ema2Source, ema2Length)

plot(ema2, title = "EMA2", color = ema2Color, force_overlay = true)

// Market Conditions
buyCondition = ta.crossover(ema1, ema2) and barstate.isconfirmed
sellCondition = ta.crossunder(ema1, ema2) and barstate.isconfirmed


// Entry Conditions
if buyCondition and strategy.position_size <= 0
    strategy.entry("BUY", direction = strategy.long)
    alert("BUY", freq = alert.freq_once_per_bar_close)

if sellCondition and strategy.position_size >= 0
    strategy.entry("SELL", direction = strategy.short)
    alert("SELL", freq = alert.freq_once_per_bar_close)
