//@version=6
indicator("Ema, RSI and High Volume breakout indicator", overlay = false)

// EMA Section
// EMA1 settings
ema1Length = input.int(defval = 9, title = "EMA1 Length", inline = "EMA1 Inline", group = "EMA1 Settings")
ema1Source = input.source(defval = close, title = "EMA1 Source", inline = "EMA1 Inline", group = "EMA1 Settings")
ema1Color = input.color(defval = color.green, title = "EMA1 Color", inline = "EMA1 Inline", group = "EMA1 Settings")
ema1 = ta.ema(ema1Source, ema1Length)
plot(ema1, title = "EMA1", color = ema1Color, force_overlay = true)

// EMA2 settings
ema2Length = input.int(defval = 26, title = "EMA2 Length", inline = "EMA2 Inline", group = "EMA2 Settings")
ema2Source = input.source(defval = close, title = "EMA2 Source", inline = "EMA2 Inline", group = "EMA2 Settings")
ema2Color = input.color(defval = color.orange, title = "EMA2 Color", inline = "EMA2 Inline", group = "EMA2 Settings")
ema2 = ta.ema(ema2Source, ema2Length)
plot(ema2, title = "EMA2", color = ema2Color, force_overlay = true)

// RSI settings
rsiSource = input.source(defval = close, title = "RSI Source", inline = "RSI Settings", group = "RSI Settings")
rsiLength = input.int(defval = 14, title = "RSI Length", inline = "RSI Settings", group = "RSI Settings")
rsiColor = input.color(defval = color.blue, title = "RSI Color", inline = "RSI Settings", group = "RSI Settings" )
rsi = ta.rsi(rsiSource, rsiLength)
upperLine = plot(70, title = "Upper Line", color = color.gray, style = plot.style_line, linestyle = plot.linestyle_dashed)
middleLine = plot(50, title = "Middle Line", color = color.gray, style = plot.style_line, linestyle = plot.linestyle_dashed)
lowerLine = plot(30, title = "Lower Line", color = color.gray, style = plot.style_line, linestyle = plot.linestyle_dashed)
plot(rsi, title = "RSI", color = rsiColor, force_overlay = false)
fill(upperLine, lowerLine, color = color.new(color.blue, 95))

// Indicator inputs
// RSI Upper Range
greenMinRsi = input.int(defval = 50, title = "Green RSI Min", inline = "green RSI", group = "Indicator Settings")
greenMaxRsi = input.int(defval = 70, title = "Green RSI Max", inline = "green RSI", group = "Indicator Settings")


// Main conditions
isGreen = ema1 > ema2 and rsi > greenMinRsi and rsi < greenMaxRsi and barstate.isconfirmed
finalColor = isGreen ? color.teal : color.red
plot(close + 10, color = finalColor, force_overlay = true)

// Volume Section
isGreen1 = close > open and barstate.isconfirmed
finalColor1 = isGreen1 ? color.teal : color.red
// plot(volume, title = "Volume", style = plot.style_columns, color = color.new(finalColor1, 60), force_overlay = false)
numberOfBars = input.int(defval = 3, title = "Number of bars", tooltip = "Number of previous candles used to calculate average volume", group = "Indicator Settings", minval = 1)
volumeMultiplier = input.int(defval = 2, title = "Volume Multiplier",tooltip = "Current volume must be this many times the average volume", group = "Indicator Settings", minval = 1)
averageVolume = ta.sma(volume[1], numberOfBars)
highVolume = volume >= averageVolume * volumeMultiplier
plotshape(highVolume, title = "High Volume", style = shape.triangleup, location = location.belowbar, color = color.teal, size = size.normal, force_overlay = true, text = "VOL", textcolor = color.white)
